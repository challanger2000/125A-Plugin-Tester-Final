  function(add_125a_qa_fixture target package_name fixture_kind)
    smtg_add_vst3plugin(${target} PACKAGE_NAME "${package_name}" SOURCES_LIST tests/qa_fixture.cpp)
    target_link_libraries(${target} PRIVATE sdk)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE QA_FIXTURE_KIND=${fixture_kind} _CRT_SECURE_NO_WARNINGS NOMINMAX)
  endfunction()
